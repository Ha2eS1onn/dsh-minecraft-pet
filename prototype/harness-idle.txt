  // 探针：把 skin 画布的非透明像素与尺寸写进 #probe，供 --dump-dom 抓取
  function probe() {
    var c = document.querySelector('canvas.dshwv-canvas')
    var out = document.getElementById('probe')
    if (!c) { out.textContent = 'no canvas'; return }
    var ctx = c.getContext('2d')
    var rows = []
    rows.push('canvas ' + c.width + 'x' + c.height + ' display=' + (c.style.display || 'css'))
    try {
      var s = window.__dshPetState ? window.__dshPetState() : null
      rows.push('engine=' + (s ? JSON.stringify(s) : 'none'))
    } catch (e) { rows.push('engine read fail') }
    try {
      var d = ctx.getImageData(0, 0, c.width, c.height).data
      var opaque = 0, minx = 1e9, miny = 1e9, maxx = -1, maxy = -1
      for (var y = 0; y < c.height; y++) {
        for (var x = 0; x < c.width; x++) {
          if (d[(y * c.width + x) * 4 + 3] > 0) {
            opaque++
            if (x < minx) minx = x
            if (x > maxx) maxx = x
            if (y < miny) miny = y
            if (y > maxy) maxy = y
          }
        }
      }
      rows.push('opaque=' + opaque)
      rows.push('bounds=' + (opaque ? [minx, miny, maxx, maxy].join(',') : 'none'))
      var h = 0
      for (var i = 0; i < d.length; i += 4) { h = (h * 16777619 ^ (d[i] << 16 ^ d[i + 1] << 8 ^ d[i + 2] ^ d[i + 3])) >>> 0 }
      rows.push('hash=' + h.toString(16))
    } catch (e) { rows.push('read fail: ' + e.message) }
    out.textContent = rows.join(' | ')
  }
  setTimeout(probe, 1500)
  setTimeout(probe, 3000)
  setTimeout(probe, 8000)
</script>
